import React, { useState, useEffect } from "react";
import { useLocation, useNavigate, useParams } from "react-router-dom";
import ReactQuill from "react-quill-new";
import "react-quill-new/dist/quill.snow.css";
import {
  ChevronLeft,
  Download,
  Save,
  CheckCircle,
  FileText,
  Printer,
} from "lucide-react";
import {
  OrangeButton,
  ExitButton,
  WhiteButton,
} from "../components/ActionButtons";
import html2pdf from "html2pdf.js";
import RoomHeader from "../components/RoomHeader";

const DocumentPreview = () => {
  const { roomNumber } = useParams();
  const location = useLocation();
  const navigate = useNavigate();

  // รับข้อมูลจากหน้า RoomDetail
  const { content, title } = location.state || {};

  const [editContent, setEditContent] = useState(content || "");
  const [isSaved, setIsSaved] = useState(true); // เช็คว่าบันทึกหรือยัง
  const [showSaveToast, setShowSaveToast] = useState(false);

  useEffect(() => {
    if (!content) {
      navigate(`/rooms/${roomNumber}`);
    }
  }, [content, navigate, roomNumber]);

  // ฟังก์ชันบันทึกร่าง (Save Draft)
  const handleSaveDraft = () => {
    console.log("Saving draft to database/localStorage...", editContent);
    // ตรงนี้สามารถยิง API ไปเซฟที่ Backend ได้

    setIsSaved(true);
    setShowSaveToast(true);
    setTimeout(() => setShowSaveToast(false), 2000); // หายไปใน 2 วินาที
  };

  // ฟังก์ชันดาวน์โหลด PDF
  const handleDownloadPDF = () => {
    const element = document.querySelector(".ql-editor");
    const opt = {
      margin: 15,
      filename: `${title}_ห้อง${roomNumber}.pdf`,
      image: { type: "jpeg", quality: 0.98 },
      html2canvas: { scale: 2, useCORS: true, letterRendering: true },
      jsPDF: { unit: "mm", format: "a4", orientation: "portrait" },
    };

    html2pdf().set(opt).from(element).save();
  };

  // เมื่อมีการพิมพ์ ให้เปลี่ยนสถานะเป็นยังไม่ได้เซฟ
  const handleChange = (value) => {
    setEditContent(value);
    if (isSaved) setIsSaved(false);
  };

  const handlePrint = () => {
    window.print(); // สั่งพิมพ์ผ่าน Browser โดยตรง
  };

  // 1. กำหนดว่าจะมีปุ่มอะไรบ้าง
const modules = {
  toolbar: [
    [{ 'header': [1, 2, 3, false] }],
    ['bold', 'italic', 'underline', 'strike'], // เพิ่ม underline ตรงนี้
    [{ 'color': [] }, { 'background': [] }],   // เพิ่มการเปลี่ยนสีข้อความ/พื้นหลัง
    [{ 'list': 'ordered' }, { 'list': 'bullet' }],
    [{ 'align': [] }],                         // เพิ่มการจัดชิดซ้าย/กลาง/ขวา
    ['clean']                                  // ปุ่มล้างรูปแบบ
  ],
};

// 2. กำหนด Format ที่อนุญาตให้ใช้
const formats = [
  'header',
  'bold', 'italic', 'underline', 'strike',
  'color', 'background',
  'list', 'bullet',
  'align'
];

  return (
    <>
      <div className="relative text-center mb-8">
        <ExitButton
          onClick={() => navigate(-1)}
          className="absolute p-2 right-0 hover:bg-gray-100 rounded-full transition-colors"
        ></ExitButton>
        <h1 className="text-3xl font-bold text-gray-800">ห้อง {roomNumber}</h1>
      </div>

      <div className="max-w-4xl mx-auto px-4 sm:px-0 space-y-6 mt-2">
        {/* Top Navigation */}
        <div className="max-w-6xl mx-auto flex justify-between items-center">
          <div className="flex items-center gap-4">
            <button
              onClick={() => navigate(-1)}
              className="p-2 hover:bg-gray-100 rounded-full transition-colors text-gray-500"
            >
              <ChevronLeft size={24} />
            </button>
            <div>
              <h1 className="text-2xl font-black text-gray-700">{title}</h1>
            </div>
          </div>

          <div className="flex items-center gap-3">
            {/* ปุ่มบันทึกร่าง */}
            <button
              onClick={handleSaveDraft}
              className={`flex items-center gap-2 px-5 py-2.5 rounded-xl font-bold text-sm transition-all ${
                isSaved
                  ? "bg-gray-100 text-gray-400 cursor-default"
                  : "bg-white text-orange-400 hover:bg-orange-100 border-2 border-orange-300 "
              }`}
            >
              <Save size={18} />
              บันทึกร่าง
            </button>

            {/* ปุ่มดาวน์โหลด */}
            <button
              onClick={() => window.print()}
              className="px-4 py-2 border border-gray-300 rounded-xl hover:bg-gray-50 flex items-center gap-2"
            >
              <Printer size={18} />
              พิมพ์ด่วน (Browser)
            </button>

            {/* ตัวเลือกที่ 2: ดาวน์โหลดไฟล์ทางการ (ใช้ html2pdf หรือ Library อื่น) */}
            <OrangeButton
              label="ดาวน์โหลดไฟล์ทางการ"
              icon={Download}
              onClick={handleDownloadPDF}
            />
          </div>
        </div>

        {/* Editor Workspace */}
        <main className="flex-1 overflow-y-auto p-2">
          <div className="max-w-full mx-auto bg-white rounded-2xl border border-gray-200 overflow-hidden flex flex-col min-h-[1100px]">
            <ReactQuill
              theme="snow"
              value={editContent}
              onChange={handleChange}
              className="document-editor flex-1 flex flex-col"
              modules={modules} // ใส่ modules ที่เราตั้งค่าไว้
        formats={formats} // ใส่ formats ที่เราตั้งค่าไว้
            />
          </div>
        </main>

        {/* Save Notification Toast */}
        {showSaveToast && (
          <div className="fixed bottom-10 left-1/2 -translate-x-1/2 bg-gray-900 text-white px-6 py-3 rounded-2xl flex items-center gap-3 shadow-2xl animate-in fade-in slide-in-from-bottom-5 duration-300 z-[100]">
            <CheckCircle size={20} className="text-green-400" />
            <span className="font-bold text-sm">บันทึกข้อมูลเรียบร้อยแล้ว</span>
          </div>
        )}

        {/* Custom Styles สำหรับจำลองหน้ากระดาษ A4 */}
        <style>{`
        /* 1. กำหนด Font สำหรับทั้งหน้าจอและตอนพิมพ์ */
        @import url('https://fonts.googleapis.com/css2?family=Sarabun:wght@400;700&display=swap');
        
        .document-editor .ql-editor {
          font-family: 'Sarabun', sans-serif !important;
          font-size: 12pt; /* ขนาดมาตรฐานเอกสารไทย */
          line-height: 1.6;
          min-height: 297mm; /* ความสูง A4 */
          padding: 25mm 25mm !important; /* ระยะขอบกระดาษมาตรฐาน */
        }

        /* 2. จัดการหน้ากระดาษตอนสั่งพิมพ์ (Print Mode) */
        @media print {
          @page {
            size: A4;
            margin: 0; /* ให้ CSS จัดการ Margin เอง */
          }
          body {
            background-color: white !important;
          }
          .ql-toolbar {
            display: none !important; /* ซ่อนแถบเครื่องมือตอนพิมพ์ */
          }
          .document-editor .ql-container.ql-snow {
            border: none !important;
          }
          .document-editor .ql-editor {
            padding: 0 !important; /* ใช้ Margin จาก @page หรือตามที่ตั้งไว้ */
            overflow: visible;
          }
        }

        /* 3. ปรับแต่ง Toolbar ให้ดูสะอาดตา (Screen Mode) */
        .document-editor .ql-toolbar {
          border: none;
          border-bottom: 1px solid #eee;
          background: #fafafa;
          position: sticky;
          top: 0;
          z-index: 10;
        }
      `}</style>
      </div>
    </>
  );
};

export default DocumentPreview;
